### 方案3：检测空响应并重试（必须）

**文件：fetch.py 修改 `fetch_url_content`**

```python
def fetch_url_content(url: str, timeout: int) -> str:
    req = Request(url, headers=build_request_headers(url))
    
    with urlopen(req, timeout=timeout) as resp:
        data = resp.read()
        
        # ✅ 新增：检测空响应，这可能表示被检测为爬虫
        if len(data) == 0 and resp.status == 200:
            # 空body + 200状态 = 反爬虫响应
            raise RuntimeError(
                f"Empty response (anti-bot detection): {url}"
            )
        
        content_type = resp.headers.get_content_charset()
        encoding = content_type or "utf-8"
        return data.decode(encoding, errors="replace")
```

### 方案4：添加HTTP 401重试（关键）

**文件：fetch.py 修改 `fetch_url_content_resilient`**

```python
import time

def fetch_url_content_resilient(
    url: str, 
    timeout: int,
    max_retries: int = 3  # ← 新参数
) -> tuple[str, str]:
    """
    发起请求，遇到HTTP 401时自动重试
    """
    last_error = None
    
    for attempt in range(max_retries):
        try:
            return fetch_url_content(url, timeout), url
        
        except HTTPError as exc:
            last_error = exc
            
            # 对HTTP 401进行重试
            if exc.code == 401 and attempt < max_retries - 1:
                wait_time = min(2 ** attempt, 10)  # 指数退避：1s, 2s, 4s
                print(f"  [RETRY] HTTP 401, waiting {wait_time}s... (attempt {attempt+1}/{max_retries})")
                time.sleep(wait_time)
                continue  # ← 重试
            
            # 5xx错误尝试fallback
            fallback_url = remove_review_true(url)
            if exc.code >= 500 and fallback_url != url:
                try:
                    return fetch_url_content(fallback_url, timeout), fallback_url
                except:
                    pass
            
            # 都失败了，继续重试或最终放弃
            if attempt < max_retries - 1:
                wait_time = min(2 ** attempt, 10)
                print(f"  [RETRY] HTTP {exc.code}, waiting {wait_time}s...")
                time.sleep(wait_time)
                continue
            
            raise last_error
    
    if last_error:
        raise last_error
```
